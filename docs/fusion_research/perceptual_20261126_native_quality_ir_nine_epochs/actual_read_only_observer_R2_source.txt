def utc():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def read(path):return json.loads(Path(path).read_text())
def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def proc(pid):
    path=Path('/proc')/str(pid)
    try:
        raw=(path/'cmdline').read_bytes();stat=(path/'stat').read_text().rsplit(')',1)[1].split()
        env=(path/'environ').read_bytes().split(b'\0')
        return dict(PID=pid,ok=True,state=stat[0],start_ticks=int(stat[19]),
            argv_sha256=hashlib.sha256(raw).hexdigest(),command=raw.rstrip(b'\0').replace(b'\0',b' ').decode(),
            CUDA_VISIBLE_DEVICES=next((v.decode().split('=',1)[1] for v in env if v.startswith(b'CUDA_VISIBLE_DEVICES=')),None))
    except OSError as error:
        return dict(PID=pid,ok=False,exists_on_recheck=path.exists(),error=repr(error),
                    transient_read_is_not_a_training_failure=True)


def last_update(path):
    if not path:return None
    try:
        with Path(path).open('rb') as stream:
            stream.seek(0,2);size=stream.tell();stream.seek(max(0,size-65536))
            lines=stream.read().decode(errors='replace').splitlines()
    except OSError:return None
    for line in reversed(lines):
        try:
            value=ast.literal_eval(line)
            if isinstance(value,dict) and 'actual_updates' in value:return value
        except (ValueError,SyntaxError):pass
    return None


def save(value,emit=False):
    temporary=LATEST.with_suffix('.partial.json')
    temporary.write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n')
    os.replace(temporary,LATEST)
    if emit:
        line=json.dumps(value,ensure_ascii=False)+'\n'
        with LOG.open('a') as stream:stream.write(line)
        print(line,end='',flush=True)


def finish(status,code,**extra):
    value=dict(row,status=status,UTC=utc(),observer_intended_exit_code=code,
               actual_numeric_exit_requires_parent_tool=True,**extra)
    save(value,True)
    raise SystemExit(code)
