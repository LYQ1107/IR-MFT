"""Original SAM union and optional original BLIP; resumable NAS paths, no GT inputs.

The fixed-caption branch does not import, construct, or run BLIP.
"""
import argparse,hashlib,json,os,pathlib,time
import numpy as np,torch
from PIL import Image
from nvml_monitor import MemoryMonitor
from queue_state import atomic_json
from resume_index import existing_paths
from recovery_support import transient_failure
R=pathlib.Path(__file__).resolve().parents[1]
SAM_SETTINGS={'backbone':'vit_b','points_per_side':32,'points_per_batch':64,'pred_iou_thresh':.88,'stability_score_thresh':.95,'crop_n_layers':0,'min_mask_region_area':0,'merge':'union of all masks, white foreground on black, reproducing process_masks.py; no GT inputs'}

def digest(path):return hashlib.sha256(pathlib.Path(path).read_bytes()).hexdigest()

def mask_valid(task,path):
    try:
        d=json.loads(pathlib.Path(path).read_text());mp=pathlib.Path(d['mask_path'])
        with Image.open(mp) as im:
            im.load();a=np.asarray(im)
            good=im.format=='PNG' and im.mode=='L' and im.size==(1920,1080) and bool(np.all((a==0)|(a==255)))
        return good and d['rgb']==task['rgb'] and d['frame']==task['frame'] and d['sequence']==task['sequence'] and d.get('seed')==20261101 and d['rgb_sha256']==digest(task['rgb']) and d['mask_sha256']==digest(mp) and all(d['sam_settings'].get(k)==v for k,v in SAM_SETTINGS.items())
    except Exception as e:
        if transient_failure(e)=='storage_unavailable':raise
        return False

def valid_aux(task,path,needs_blip=True,fixed_caption=None):
    if not mask_valid(task,path):return False
    d=json.loads(pathlib.Path(path).read_text())
    return (d.get('caption_source')=='BLIP') if needs_blip else (d.get('caption_source')=='fixed' and d['caption']==fixed_caption)

def main():
    ap=argparse.ArgumentParser();ap.add_argument('--plan',required=True);a=ap.parse_args()
    plan=json.loads(pathlib.Path(a.plan).read_text());fixed=plan.get('fixed_caption');pending=[]
    candidates=existing_paths(task['output'] for task in plan['tasks'])
    for task in plan['tasks']:
        reuse=task['output'] in candidates and mask_valid(task,task['output'])
        complete=False
        if reuse:
            d=json.loads(pathlib.Path(task['output']).read_text())
            complete=d.get('caption_source')=='BLIP' if fixed is None else (d.get('caption_source')=='fixed' and d['caption']==fixed)
        if not complete:pending.append((task,reuse))
    print('AUX_RESUME',len(plan['tasks'])-len(pending),'valid skipped',len(pending),'pending','BLIP_ENABLED',fixed is None,flush=True)
    if not pending:return
    torch.set_num_threads(4);torch.backends.cuda.matmul.allow_tf32=True;torch.backends.cudnn.allow_tf32=True
    torch.cuda.set_per_process_memory_fraction(.65)
    oldload=torch.load
    def trusted_load(*args,**kw):kw.setdefault('weights_only',False);return oldload(*args,**kw)
    torch.load=trusted_load;t=time.monotonic();generator=None;blip=None
    if any(not exists for _,exists in pending):
        from segment_anything import sam_model_registry,SamAutomaticMaskGenerator
        sam=sam_model_registry['vit_b'](checkpoint=str(R/'cache/ram/SAM/sam_vit_b_01ec64.pth')).cuda().eval()
        generator=SamAutomaticMaskGenerator(sam,points_per_side=32,points_per_batch=64,pred_iou_thresh=.88,stability_score_thresh=.95,crop_n_layers=0,min_mask_region_area=0)
    if fixed is None:
        from transformers import BertTokenizer
        original=BertTokenizer.from_pretrained
        BertTokenizer.from_pretrained=lambda *args,**kw:original(str(R/'cache/ram/bert-base-uncased'),**kw)
        from blip_models.blip import blip_decoder
        from infer import load_demo_image
        blip=blip_decoder(pretrained=str(R/'cache/ram/BLIP/model_base_caption_capfilt_large.pth'),med_config=str(R/'DiffV2IR/configs/med_config.json'),image_size=384,vit='base').cuda().eval()
    torch.load=oldload;torch.cuda.synchronize();print('AUX_MODELS_LOADED',time.monotonic()-t,flush=True);mon=MemoryMonitor()
    for i,(task,reuse) in enumerate(pending):
        dest=pathlib.Path(task['output']);maskpath=pathlib.Path(task['mask_output']);seed=task.get('seed',20261101)
        torch.manual_seed(seed);torch.cuda.manual_seed_all(seed);np.random.seed(seed);torch.cuda.reset_peak_memory_stats();torch.cuda.synchronize();mon.start()
        if reuse:
            rec=json.loads(dest.read_text());samseconds=0.;rec['sam_reused_from_validated_mask']=True
        else:
            rgb=np.asarray(Image.open(task['rgb']).convert('RGB'));t=time.monotonic()
            with torch.inference_mode():masks=generator.generate(rgb)
            torch.cuda.synchronize();samseconds=time.monotonic()-t;union=np.zeros(rgb.shape[:2],bool)
            for m in masks:union|=m['segmentation']
            maskpath.parent.mkdir(parents=True,exist_ok=True);tmp=maskpath.with_name('.'+maskpath.name+f'.{os.getpid()}.tmp')
            Image.fromarray((union*255).astype(np.uint8)).save(tmp,format='PNG');os.replace(tmp,maskpath)
            rec={'sequence':task['sequence'],'frame':task['frame'],'rgb':task['rgb'],'seed':seed,'mask_path':str(maskpath),'mask_count':len(masks),'mask_union_fraction':float(union.mean()),'sam_settings':SAM_SETTINGS,'rgb_sha256':digest(task['rgb']),'mask_sha256':digest(maskpath),'sam_seconds':samseconds,'sam_original_seconds':samseconds}
        torch.manual_seed(seed);torch.cuda.manual_seed_all(seed);tcap=time.monotonic()
        if fixed is not None:caption=fixed;blipseconds=0.
        else:
            with torch.inference_mode():caption=blip.generate(load_demo_image(384,'cuda',task['rgb']).cuda(),sample=True,top_p=.9,max_length=20,min_length=5)[0]
            torch.cuda.synchronize();blipseconds=time.monotonic()-tcap
        nv=mon.stop();rec.update(caption=caption,edit='turn the visible image of '+caption+' into infrared',caption_source='fixed' if fixed is not None else 'BLIP',sam_seconds=samseconds,blip_seconds=blipseconds,seconds=samseconds+blipseconds,first_after_load=i==0,peak_allocated_bytes=torch.cuda.max_memory_allocated(),peak_reserved_bytes=torch.cuda.max_memory_reserved(),nvml_peak_bytes=nv,gpu=torch.cuda.get_device_name(),physical_gpu=int(os.environ['PROBE_NVML_PHYSICAL_GPU']),completed_unix=time.time())
        atomic_json(dest,rec)
        with (R/'evidence'/f'{plan["experiment"]}_records.jsonl').open('a') as out:out.write(json.dumps(rec)+'\n')
        print('AUX_FRAME',i+1,'/',len(pending),task['sequence'],task['frame'],'sam_reused',reuse,'caption',caption,'seconds',round(rec['seconds'],3),flush=True)
if __name__=='__main__':main()
