from types import SimpleNamespace
import numpy as np
import pytest
from tools.mft_candidate_risk_features import candidate_features,FEATURE_NAMES,IR_FEATURES,rgb_only_mask
from tools.mft_candidate_risk_labels import current_detection_labels,PastLabelHistory,edge_targets


def unit(x):
    v=np.asarray(x,np.float32);return v/np.linalg.norm(v)


def track(rgb,ir=(1,0),identity=1):
    return SimpleNamespace(accepted=5,rgb=unit(rgb),ir=unit(ir),last_box=np.asarray([0,0,20,20],np.float32),
        tlbr=np.asarray([1,0,21,20],np.float32),frame_id=4,memory_frame=3,start_frame=1,
        memory_quality=.8,mean=np.asarray([11,10,1,20,1,0,0,0],np.float64),track_id=identity)


def detection(rgb,ir=(1,0),available=True,box=(1,0,21,20)):
    return SimpleNamespace(tlbr=np.asarray(box,np.float32),score=.9,
        observation=dict(box=np.asarray(box,np.float32),rgb=unit(rgb),ir=unit(ir),quality=.9,ir_available=available))


def test_features_ignore_identity_names_and_training_label_changes():
    tracks=[track((1,0),identity=1),track((0,1),identity=2)]
    dets=[detection((1,0)),detection((0,1))];base=np.full((2,2),.3)
    a,allowed=candidate_features(tracks,dets,base,5,.13137420405549355)
    assert allowed.all()and a.shape==(2,2,len(FEATURE_NAMES))and np.isfinite(a).all()
    for t in tracks:t.track_id+=1000
    b,_=candidate_features(tracks,dets,base,5,.13137420405549355)
    np.testing.assert_array_equal(a,b)
    y,_=edge_targets([1,2],np.asarray([1,2]),allowed)
    z,_=edge_targets([22,11],np.asarray([11,22]),allowed)
    assert not np.array_equal(y,z)
    np.testing.assert_array_equal(a,candidate_features(tracks,dets,base,5,.13137420405549355)[0])


def test_missing_ir_is_unreadable_and_exact_rgb_mask():
    class Unreadable:
        def __array__(self,*args,**kwargs):raise AssertionError('Unavailable IR read')
    t=track((1,0));d=detection((1,0),available=False)
    t.ir=Unreadable();d.observation['ir']=Unreadable()
    a,_=candidate_features([t],[d],np.asarray([[.3]]),5,.13137420405549355)
    assert (a[...,IR_FEATURES]==0).all()
    reference,_=candidate_features([track((1,0))],[detection((1,0))],np.asarray([[.3]]),5,.13137420405549355)
    np.testing.assert_array_equal(rgb_only_mask(a),rgb_only_mask(reference))


def test_weak_competitors_stay_in_rgb_uniqueness_margin():
    a=unit([.4,np.sqrt(1-.4**2)]);b=unit([.35,np.sqrt(1-.35**2)])
    x,eligible=candidate_features([track((1,0))],[detection(a),detection(b)],np.asarray([[.3,.4]]),5,.13137420405549355)
    assert eligible.all()
    assert abs(float(x[0,0,FEATURE_NAMES.index('RGB_row_margin')])-.05)<1e-6
    assert abs(float(x[0,1,FEATURE_NAMES.index('RGB_row_margin')])+.05)<1e-6


def test_current_gt_labels_respect_unknown_ignore_and_frame_only_assignment():
    d=np.asarray([[0,0,20,20,.9],[40,0,60,20,.9],[100,0,120,20,.9],[200,0,220,20,.9],[0,0,20,6,.9]],np.float32)
    gt=np.asarray([[1,7,0,0,20,20,1,1,1],[1,8,40,0,20,20,1,1,1],[1,9,100,0,20,20,0,2,1]],np.float64)
    np.testing.assert_array_equal(current_detection_labels(d,gt,frame=1,split='train'),[7,8,-1,0,-1])
    y,known=edge_targets([7,None],np.asarray([7,8,-1,0]),np.ones((2,4),bool))
    np.testing.assert_array_equal(known,[[True,True,False,True],[False,False,False,False]])
    assert y[0,0]==1 and y[0,1]==y[0,3]==0


def test_past_history_cannot_peek_at_current_or_another_sequence():
    h=PastLabelHistory()
    for frame in range(1,4):
        assert h.prior(1,frame)is None
        h.commit_after_update(frame,{1:7})
    assert h.prior(1,4)==7
    with pytest.raises(AssertionError):h.prior(1,3)
    h.commit_after_update(4,{1:8})
    assert h.prior(1,5)is None  #3/4 purity below0.8
    assert PastLabelHistory().prior(1,1)is None


def test_label_interface_refuses_other_splits_and_future_gt_frames():
    d=np.asarray([[0,0,20,20,.9]],np.float32)
    gt=np.asarray([[2,7,0,0,20,20,1,1,1]],np.float64)
    with pytest.raises(AssertionError):current_detection_labels(d,gt,frame=1,split='train')
    for split in ['dev','test']:
        with pytest.raises(AssertionError):current_detection_labels(d,gt,frame=2,split=split)
