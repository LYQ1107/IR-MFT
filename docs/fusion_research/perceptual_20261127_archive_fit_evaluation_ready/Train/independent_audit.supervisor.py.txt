import datetime
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

root=Path('/data3/liuyeqiang/IR-MFT')
base=Path('/var/tmp/liuyeqiang_ir_mft_perceptual_20261122/archive_reasoning_20261127')
mode='run';name='independent_audit_r1'
source=root/'tools/audit_mft_archive_fit.py';command=[sys.executable,'-u',str(source),mode]
assert os.environ.get('CUDA_VISIBLE_DEVICES')==''
target=base/(name+'.execution.json');log=base/(name+'.stdout.txt')
assert not target.exists()and not log.exists()
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def utc():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def save(value):
    temporary=target.with_suffix('.tmp.json');temporary.write_text(json.dumps(value,indent=2)+'\n');os.replace(temporary,target)
sources={n:sha(root/n)for n in ['tools/audit_mft_archive_fit.py','tests/test_mft_archive_fit_audit.py']}
started=time.monotonic();snapshots=[]
with log.open('xb',buffering=0)as stream:
    p=subprocess.Popen(command,cwd=root,stdin=subprocess.DEVNULL,stdout=stream,stderr=subprocess.STDOUT,start_new_session=True)
    expected=b'\0'.join(os.fsencode(s)for s in command)+b'\0'
    for _ in range(100):
        proc=Path('/proc')/str(p.pid);raw=(proc/'cmdline').read_bytes()
        snapshots.append(dict(bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest()))
        if raw==expected:
            fields=(proc/'stat').read_text().rsplit(') ',1)[1].split()
            visible=next((v.decode().split('=',1)[1]for v in(proc/'environ').read_bytes().split(b'\0')if v.startswith(b'CUDA_VISIBLE_DEVICES=')),None)
            assert visible==''
            actual=dict(PID=p.pid,start_ticks=int(fields[19]),raw_argv_sha256=hashlib.sha256(raw).hexdigest(),CUDA_VISIBLE_DEVICES=visible)
            break
        time.sleep(.01)
    else:raise RuntimeError('Preserve actual child; argv was not observed')
    value=dict(status='ACTUAL_ARCHIVE_CPU_WORKER_RUNNING',mode=mode,UTC_started=utc(),command=command,
        source_hashes=sources,supervisor_sha256=sha(Path(__file__)),actual_child_exit_code=None,
        identity=actual,startup_argv_snapshots=snapshots,log=str(log),GPU_work=False,new_accuracy=False)
    save(value);print(json.dumps(value),flush=True)
    code=p.wait()
value.update(status='CLOSED_ACTUAL_ARCHIVE_CPU_WORKER',actual_child_exit_code=code,UTC_finished=utc(),
    runtime_seconds=time.monotonic()-started,source_unchanged=all(sha(root/n)==h for n,h in sources.items()),
    stdout_sha256=sha(log))
save(value);print(json.dumps(value),flush=True)
raise SystemExit(code if value['source_unchanged']else 1)
